<script>
  let timerActive = false;
  let timerInterval = null;
  let secondsRemaining = 25 * 60;

  function toggleSubtask(rowElement) {
    const box = rowElement.querySelector('.check-box');
    const text = rowElement.querySelector('.task-text');
    const isChecked = box.classList.contains('bg-secondary');

    if (isChecked) {
      box.classList.remove('bg-secondary', 'text-on-secondary');
      box.classList.add('bg-surface-container-high', 'text-transparent');
      text.classList.remove('line-through', 'text-outline');
      text.classList.add('text-on-surface');
    } else {
      box.classList.remove('bg-surface-container-high', 'text-transparent');
      box.classList.add('bg-secondary', 'text-on-secondary');
      text.classList.remove('text-on-surface');
      text.classList.add('line-through', 'text-outline');
    }
    updateProgressCounter();
  }

  function updateProgressCounter() {
    const total = document.querySelectorAll('.subtask-item').length;
    const completed = document.querySelectorAll('.subtask-item .bg-secondary').length;
    const counter = document.getElementById('subtask-counter');
    if (counter) {
      counter.textContent = `${completed} of ${total} completed`;
    }
  }

  function showAddSubtask() {
    document.getElementById('add-subtask-container').classList.remove('hidden');
    document.getElementById('add-subtask-trigger').classList.add('hidden');
    document.getElementById('new-subtask-input').focus();
  }

  function confirmAddSubtask() {
    const input = document.getElementById('new-subtask-input');
    const textVal = input.value.trim();
    if (!textVal) return;

    const list = document.getElementById('subtask-list');
    const newRow = document.createElement('div');
    newRow.className = 'subtask-item group flex items-start gap-3 p-2.5 rounded-lg transition-colors hover:bg-surface-container-low cursor-pointer';
    newRow.onclick = function() { toggleSubtask(newRow); };
    newRow.innerHTML = `
      <button type="button" class="check-box mt-0.5 w-5 h-5 rounded-full flex items-center justify-center shrink-0 transition-transform active:scale-90 bg-surface-container-high text-transparent shadow-sm" aria-label="Toggle subtask">
        <span class="material-symbols-outlined text-[15px]">check</span>
      </button>
      <div class="flex-1 min-w-0 flex items-center justify-between gap-2">
        <span class="task-text font-body-md text-body-md text-on-surface truncate">
          ${textVal}
        </span>
        <span class="font-label-sm text-label-sm text-on-surface-variant bg-surface-container px-2 py-0.5 rounded-full shrink-0">New</span>
      </div>
    `;

    list.appendChild(newRow);
    input.value = '';
    document.getElementById('add-subtask-container').classList.add('hidden');
    document.getElementById('add-subtask-trigger').classList.remove('hidden');
    updateProgressCounter();
  }

  function toggleTimer() {
    const btnIcon = document.getElementById('timer-icon');
    const btnLabel = document.getElementById('timer-label');
    const display = document.getElementById('timer-display');

    if (!timerActive) {
      timerActive = true;
      btnIcon.textContent = 'pause';
      btnLabel.textContent = 'Pause Focus';
      timerInterval = setInterval(() => {
        if (secondsRemaining > 0) {
          secondsRemaining--;
          const m = Math.floor(secondsRemaining / 60).toString().padStart(2, '0');
          const s = (secondsRemaining % 60).toString().padStart(2, '0');
          display.textContent = `${m}:${s}`;
        } else {
          clearInterval(timerInterval);
          btnIcon.textContent = 'replay';
          btnLabel.textContent = 'Restart';
          timerActive = false;
        }
      }, 1000);
    } else {
      timerActive = false;
      clearInterval(timerInterval);
      btnIcon.textContent = 'play_arrow';
      btnLabel.textContent = 'Resume Focus Block';
    }
  }

  function completeMainTask() {
    const btn = document.getElementById('complete-main-btn');
    const text = document.getElementById('complete-btn-text');
    btn.classList.toggle('bg-secondary');
    btn.classList.toggle('bg-primary');
    if (btn.classList.contains('bg-secondary')) {
      text.textContent = 'Completed!';
    } else {
      text.textContent = 'Mark Task as Complete';
    }
  }

  function confirmDelete() {
    if (confirm('Are you sure you want to remove this task?')) {
      history.back();
    }
  }
</script></main></body></html>
